    </>
  );
}

      Install
      
Start in one minute.

      
Every setup scans your files and writes only the classes, color tokens and fonts you use.

    

    

      

        
        
        
        
        
        
      

      
npm install -D @birdycom/omniacss-cli
npx omniacss init
npx omniacss build --watch      # while you work
npx omniacss build --minify     # for production

      
// postcss.config.js
import omniacss from "@birdycom/omniacss/postcss";
export default {
  plugins: [omniacss({ content: ["index.html", "src/**/*.{js,ts,jsx,tsx,vue,svelte}"] })],
};
/* src/style.css */
@omnia;

      
// postcss.config.js (Laravel + Vite)
import omniacss from "@birdycom/omniacss/postcss";
export default {
  plugins: [omniacss({ content: ["resources/views/**/*.blade.php", "resources/js/**/*.{js,vue,tsx}"] })],
};
/* resources/css/app.css */
@omnia;

      
npm install -D @birdycom/omniaui
npx omniaui list
npx omniaui add modal
npx omniaui add cms-dashboard --to resources/views/admin

      
# OmniaCSS IntelliSense: utilities and OmniaUI classes
code --install-extension Birdycom.omniacss-intellisense     # VS Code
cursor --install-extension Birdycom.omniacss-intellisense   # Cursor (Open VSX)
